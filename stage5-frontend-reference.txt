function Dashboard({ onSignedOut }: { onSignedOut: () => void }) {

  const [showCreateForm, setShowCreateForm] = useState(false)

  useEffect(() => {
    if (home.error instanceof SessionExpiredError) {
      onSignedOut()
    }
  }, [home.error, onSignedOut])

 if (home.isPending) {
  return <DashboardSkeleton />
}

  if (home.isError) {

      return null

          Try again

        >

        </button>
      )}
    </div>

    {showCreateForm && (
      <div className="mb-6">
        <CreateStokvelForm
          onCancel={() => setShowCreateForm(false)}
          onCreated={() => setShowCreateForm(false)}

            key={stokvel.stokvelId}
            stokvel={stokvel}

      </div>
    )}
  </section>
)
}

function App() {
  const [authState, setAuthState] = useState<
    'checking' | 'signed-in' | 'signed-out'
  >('checking')

  useEffect(() => {
    let active = true

    async function restoreSession() {
      const token = getAccessToken() ?? await refreshAccessToken()

      if (active) {
        setAuthState(token ? 'signed-in' : 'signed-out')
      }
    }

    void restoreSession()

    return () => {
      active = false
    }
  }, [])

  function handleSignedOut() {
    queryClient.clear()
    setAuthState('signed-out')
  }

  async function handleLogout() {
    queryClient.clear()
    setAuthState('signed-out')
    await logout()
  }

  function handleLogin() {
    queryClient.clear()
    setAuthState('signed-in')
  }

  return (
    <main className="app-container space-y-10">
      <header className="page-header">
        <div className="flex flex-wrap items-start justify-between gap-4">
          <div>
            <h1 className="text-3xl font-bold">
              RondiTrack Dashboard
            </h1>
            <p className="mt-2 text-muted-foreground">
              Assignment 7.3 â€” Live RondiTrack Data
            </p>
          </div>

          <div className="flex items-center gap-3">
            {authState === 'signed-in' && (
              <button
                type="button"
                onClick={() => void handleLogout()}
                className="rounded-md border px-4 py-2"
              >
                Log out
              </button>
            )}
            <ThemeToggle />
          </div>
        </div>
      </header>

      {authState === 'checking' ? (
        <p role="status">Checking your session...</p>
      ) : authState === 'signed-out' ? (
        <LoginForm onLogin={handleLogin} />
      ) : (
        <Dashboard onSignedOut={handleSignedOut} />
      )}
    </main>
  )
}

export default App

/**
 * Access tokens are kept only in memory.
 *
 * Single-flight refresh means concurrent requests share one
 * refresh promise rather than rotating the cookie repeatedly.
 *
 * RondiTrack rotates refresh cookies and can revoke a token
 * family when an already-used refresh token is presented.
 */

const API_BASE_URL =
  import.meta.env.VITE_API_BASE_URL ?? "";

let accessToken: string | null = null;
let inFlightRefresh: Promise<string | null> | null = null;
let sessionVersion = 0;
type TokenResponse = {
  accessToken: string;
  tokenType: string;
  expiresIn: number;
};

export class SessionExpiredError extends Error {
  constructor() {
    super("Your session has expired. Please sign in again.");
    this.name = "SessionExpiredError";
  }
}

export function getAccessToken(): string | null {
  return accessToken;
}

function clearSession(): void {
  sessionVersion++;
  accessToken = null;
}

function apiUrl(path: string): string {
  return `${API_BASE_URL.replace(/\/$/, "")}${path}`;
}

export async function login(
  email: string,
  password: string,
): Promise<void> {
  clearSession();

  const response = await fetch(apiUrl("/api/v1/auth/login"), {
    method: "POST",
    credentials: "include",
    headers: {
      "Content-Type": "application/json",
    },
    body: JSON.stringify({ email, password }),
  });

  if (!response.ok) {
    throw new Error(
      response.status === 401
        ? "The email address or password is incorrect."
        : "Unable to sign in. Please try again.",
    );
  }

  const data = (await response.json()) as TokenResponse;
  accessToken = data.accessToken;
}


async function doRefresh(): Promise<string | null> {
  const versionAtStart = sessionVersion;

  try {
    const response = await fetch(apiUrl("/api/v1/auth/refresh"), {
      method: "POST",
      credentials: "include",
    });

    if (versionAtStart !== sessionVersion) {
      return null;
    }

    if (!response.ok) {
      clearSession();
      return null;
    }

    const data = (await response.json()) as TokenResponse;

    if (versionAtStart !== sessionVersion) {
      return null;
    }

    accessToken = data.accessToken;
    return accessToken;
  } catch {
    if (versionAtStart === sessionVersion) {
      clearSession();
    }

    return null;
  }
}

export function refreshAccessToken(): Promise<string | null> {
  if (inFlightRefresh) {
    return inFlightRefresh;
  }

  const refreshPromise = doRefresh();
  inFlightRefresh = refreshPromise;

  void refreshPromise.finally(() => {
    if (inFlightRefresh === refreshPromise) {
      inFlightRefresh = null;
    }
  });

  return refreshPromise;
}

export async function logout(): Promise<void> {
  clearSession();

  try {
    await fetch(apiUrl("/api/v1/auth/logout"), {
      method: "POST",
      credentials: "include",
    });
  } catch {
    // The browser must remain signed out even if offline.
  }
}

/**
 * Authenticated fetch adapter for openapi-fetch.
 * A request is retried at most once after a 401.
 */
export async function authFetch(
  input: RequestInfo | URL,
  init?: RequestInit,
): Promise<Response> {
const request = new Request(input, {
  ...init,
  credentials: "include",
});  const retryableRequest = request.clone();

  let token = accessToken;

  if (!token) {
    token = await refreshAccessToken();
    if (!token) {
      throw new SessionExpiredError();
    }
  }

  const firstRequest = request.clone();
  firstRequest.headers.set("Authorization", `Bearer ${token}`);

  const response = await fetch(firstRequest);

  if (response.status !== 401) {
    return response;
  }

  // Another request may already have refreshed the token.
  const nextToken =
    accessToken && accessToken !== token
      ? accessToken
      : await refreshAccessToken();

  if (!nextToken) {
    throw new SessionExpiredError();
  }

  retryableRequest.headers.set(
    "Authorization",
    `Bearer ${nextToken}`,
  );

  const retryResponse = await fetch(retryableRequest);

  if (retryResponse.status === 401) {
    clearSession();
    throw new SessionExpiredError();
  }

  return retryResponse;
}

import { useState } from 'react'
import { ChevronDown, ChevronUp } from 'lucide-react'
import { DueCountdown } from '@/components/DueCountdown'
import type {
    ContributionCycle,
    ContributionItem,
} from '@/api/types'

import { Button } from '@/components/ui/button'
import {
    Card,
    CardContent,
    CardHeader,
} from '@/components/ui/card'
import { ContributionCard } from '@/components/ContributionCard'
import type {
    DemoContributionStatus,
} from '@/components/ContributionCard'
import { StatusBadge } from '@/components/StatusBadge'

import {
    groupContributionsByCycle,
} from '@/lib/groupContributions'

import { formatCents } from '@/lib/money'
import { toSafeCents } from '@/lib/apiMoney'
import { formatSastDateTime } from '@/lib/dates'

type ContributionListProps = {
    cycles: readonly ContributionCycle[]
    contributions: readonly ContributionItem[]
    demoStatuses?: Readonly<Record<string, DemoContributionStatus>>
}

export function ContributionList({
    cycles,
    contributions,
    demoStatuses = {},
}: ContributionListProps) {
    const groups = groupContributionsByCycle(
        cycles,
        contributions,
    )

    // Store only user overrides, not the grouped data.
    const [expandedOverrides, setExpandedOverrides] =
        useState<Record<string, boolean>>({})

    function isExpanded(id: string, status?: string): boolean {
        const override = expandedOverrides[id]

        if (override !== undefined) {
            return override
        }

        return status?.trim().toLowerCase() === 'open'
    }

    const allExpanded =
        groups.length > 0 &&
        groups.every((group) =>
            isExpanded(group.id, group.cycle?.status),
        )

    function toggleAll() {
        const nextExpanded = !allExpanded

        setExpandedOverrides(
            Object.fromEntries(
                groups.map((group) => [group.id, nextExpanded]),
            ),
        )
    }

    return (
        <div className="space-y-4">
            <div className="flex justify-end">
                <Button
                    type="button"
                    variant="outline"
                    onClick={toggleAll}
                    disabled={groups.length === 0}
                >
                    {allExpanded ? 'Collapse all' : 'Expand all'}
                </Button>
            </div>

            {groups.length === 0 && (
                <p className="text-muted-foreground">
                    No contribution cycles yet.
                </p>
            )}

            {groups.map((group) => {
                const cycle = group.cycle
                const expanded = isExpanded(
                    group.id,
                    cycle?.status,
                )

                const headingId = `cycle-heading-${group.id}`
                const contentId = `cycle-content-${group.id}`

                return (
                    <Card
                        key={group.id}
                        className="min-w-0"
                        aria-labelledby={headingId}
                    >
                        <CardHeader>
                            <div className="flex flex-wrap items-start justify-between gap-3">
                                <div className="min-w-0 space-y-2">
                                    <h3
                                        id={headingId}
                                        className="break-words text-lg font-semibold"
                                    >
                                        {cycle
                                            ? `Cycle: ${cycle.period}`
                                            : 'Unknown cycle'}
                                    </h3>

                                    {cycle && (
                                        <div className="flex flex-wrap items-center gap-2">
                                            <StatusBadge status={cycle.status ?? 'Unknown'} />

                                            <DueCountdown
                                                dueAtUtc={cycle.dueAtUtc}
                                                status={cycle.status ?? 'Unknown'}
                                            />
                                        </div>
                                    )}
                                </div>

                                <Button
                                    type="button"
                                    variant="outline"
                                    aria-expanded={expanded}
                                    aria-controls={contentId}
                                    onClick={() =>
                                        setExpandedOverrides((previous) => ({
                                            ...previous,
                                            [group.id]: !expanded,
                                        }))
                                    }
                                >
                                    {expanded ? (
                                        <ChevronUp aria-hidden="true" />
                                    ) : (
                                        <ChevronDown aria-hidden="true" />
                                    )}
                                    {expanded ? 'Collapse' : 'Expand'}
                                </Button>
                            </div>

                            {cycle && (
                                <div className="grid gap-2 pt-2 text-sm sm:grid-cols-3">
                                    <p>
                                        <span className="text-muted-foreground">
                                            Due:
                                        </span>{' '}
                                        {formatSastDateTime(cycle.dueAtUtc)}
                                    </p>

                                    <p>
                                        <span className="text-muted-foreground">
                                            Collected:
                                        </span>{' '}
                                        {cycle.totalCollectedCents == null
                                            ? 'Not available'
                                            : formatCents(
                                                toSafeCents(
                                                    cycle.totalCollectedCents,
                                                ),
                                            )}
                                    </p>

                                    <p>
                                        <span className="text-muted-foreground">
                                            Amount due:
                                        </span>{' '}
                                        {cycle.amountDueCents == null
                                            ? 'Not available'
                                            : formatCents(
                                                toSafeCents(
                                                    cycle.amountDueCents,
                                                ),
                                            )}
                                    </p>
                                </div>
                            )}
                        </CardHeader>

                        <CardContent>
                            <div id={contentId} hidden={!expanded}>
                                {group.contributions.length === 0 ? (
                                    <p className="text-sm text-muted-foreground">
                                        No contributions yet.
                                    </p>
                                ) : (
                                    <div className="grid gap-3 md:grid-cols-2">
                                        {group.contributions.map(
                                            (contribution) => (
                                                <ContributionCard
                                                    key={contribution.id}
                                                    contribution={contribution}
                                                    demoStatus={
                                                        demoStatuses[contribution.id]
                                                    }
                                                />
                                            ),
                                        )}
                                    </div>
                                )}
                            </div>
                        </CardContent>
                    </Card>
                )
            })}
        </div>
    )
}

import { useState, type FormEvent } from 'react'
import { useCreateStokvel } from '@/api/queries'

type Props = {
  onCancel: () => void
  onCreated: () => void
}

export function CreateStokvelForm({
  onCancel,
  onCreated,
}: Props) {
  const [name, setName] = useState('')
  const [amount, setAmount] = useState('500')
  const createStokvel = useCreateStokvel()

  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault()

    const contributionAmount = Number(amount)
    const trimmedName = name.trim()

    if (
      trimmedName.length === 0 ||
      trimmedName.length > 100 ||
      !Number.isFinite(contributionAmount) ||
      contributionAmount <= 0
    ) {
      return
    }

    try {
      await createStokvel.mutateAsync({
        name: trimmedName,
        contributionAmount,
      })

      onCreated()
    } catch {
      // The mutation error is displayed below.
    }
  }

  return (
    <form
      onSubmit={(event) => void handleSubmit(event)}
      className="space-y-4 rounded-lg border p-5"
    >
      <h3 className="text-xl font-semibold">
        Create a New Stokvel
      </h3>

      <div className="space-y-2">
        <label htmlFor="stokvel-name" className="block font-medium">
          Stokvel name
        </label>

        <input
          id="stokvel-name"
          type="text"
          value={name}
          onChange={(event) => setName(event.target.value)}
          maxLength={100}
          required
          disabled={createStokvel.isPending}
          placeholder="e.g. Ubuntu Savings Club"
          className="w-full rounded-md border bg-background px-3 py-2"
        />
      </div>

      <div className="space-y-2">
        <label htmlFor="contribution-amount" className="block font-medium">
          Contribution amount (R)
        </label>

        <input
          id="contribution-amount"
          type="number"
          min="0.01"
          step="0.01"
          value={amount}
          onChange={(event) => setAmount(event.target.value)}
          required
          disabled={createStokvel.isPending}
          className="w-full rounded-md border bg-background px-3 py-2"
        />
      </div>

      {createStokvel.isError && (
        <p role="alert" className="text-sm text-red-600">
          {createStokvel.error.message}
        </p>
      )}

      <div className="flex gap-3">
        <button
          type="submit"
          disabled={createStokvel.isPending}
          className="rounded-md bg-primary px-4 py-2 text-primary-foreground disabled:opacity-50"
        >
          {createStokvel.isPending
            ? 'Creating...'
            : 'Create Stokvel'}
        </button>

        <button
          type="button"
          onClick={onCancel}
          disabled={createStokvel.isPending}
          className="rounded-md border px-4 py-2"
        >
          Cancel
        </button>
      </div>
    </form>
  )
}
